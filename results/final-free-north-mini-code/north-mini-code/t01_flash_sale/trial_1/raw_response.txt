def purchase(db, product_id: int, qty: int) -> bool:
    db.begin(isolation="repeatable read")
    # Lock the product row and read current stock
    rows = db.execute(
        "SELECT stock FROM products WHERE id = %(pid)s FOR UPDATE",
        {"pid": product_id}
    )
    if not rows:
        # Product does not exist (should not happen per problem statement)
        return False
    stock = rows[0][0]
    if stock >= qty:
        db.execute(
            "UPDATE products SET stock = stock - %(qty)s WHERE id = %(pid)s",
            {"qty": qty, "pid": product_id}
        )
        db.execute(
            "INSERT INTO orders(product_id, qty) VALUES (%(pid)s, %(qty)s)",
            {"pid": product_id, "qty": qty}
        )
        db.commit()
        return True
    # Not enough stock – transaction will be rolled back automatically
    return False
