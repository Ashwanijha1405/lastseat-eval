import psycopg.errors

def approve_expense(db, expense_id: int) -> bool:
    max_attempts = 5
    for _ in range(max_attempts):
        try:
            db.begin(isolation="repeatable read")

            # Lock the expense row to ensure exclusive access
            expense = db.execute(
                "SELECT department_id, amount, status FROM expenses WHERE id = %(id)s FOR UPDATE",
                {"id": expense_id}
            )
            if not expense:
                db.rollback()
                return False
            department_id, amount, status = expense[0]
            if status != "pending":
                db.rollback()
                return False

            # Lock the department row to protect its budget
            dept = db.execute(
                "SELECT budget FROM departments WHERE id = %(dept_id)s FOR UPDATE",
                {"dept_id": department_id}
            )
            if not dept:
                db.rollback()
                return False
            budget = dept[0][0]

            # Compute current total of approved expenses for this department
            total = db.execute(
                "SELECT COALESCE(SUM(amount), 0) FROM expenses "
                "WHERE department_id = %(dept_id)s AND status = 'approved'",
                {"dept_id": department_id}
            )[0][0] or 0

            # Check budget constraint
            if total + amount > budget:
                db.rollback()
                return False

            # Perform the approval
            db.execute(
                "UPDATE expenses SET status = 'approved' WHERE id = %(id)s",
                {"id": expense_id}
            )
            db.commit()
            return True

        except psycopg.errors.Error:
            # Roll back any open transaction and retry
            try:
                db.rollback()
            except Exception:
                pass
            continue

    # If we exhausted retries, raise the last error (or simply re‑raise)
    raise
