Every operation that changes a room takes a lock on that room's `rooms` row first. That makes the check-then-write sequence atomic per room, and it cannot deadlock.

**Why this is correct under concurrency**

- **One lock per room.** `book` and `reschedule` begin by running `SELECT ... FROM rooms WHERE id = ... FOR NO KEY UPDATE` on the room they affect. Requests for the same room run their checks one at a time. Requests for different rooms never block each other.
- **Fresh reads after the lock.** The isolation level is READ COMMITTED, so each statement gets a new snapshot. Once the lock is held, the overlap check and the per-day count see every committed change from earlier lock holders. The rules are checked against the true state, so overlaps and a 4th booking on a day cannot happen.
- **No deadlocks.** A request locks only one room, so there is no lock cycle. A booking never changes room, so `reschedule` can read the booking's `room_id` safely before locking.
- **Cancelled during a reschedule.** After taking the room lock, `reschedule` re-reads the booking with `FOR UPDATE`. If a concurrent `cancel` deleted it, `reschedule` returns `False`.
- **No conflict with the foreign key check.** The lock is `FOR NO KEY UPDATE`, which does not conflict with the `FOR KEY SHARE` lock that the FK check on `bookings` inserts takes. It does conflict with itself, which is the per-room serialization we want.
- **`cancel` is a single statement.** It is an atomic `DELETE ... RETURNING`.
- **Requests that fit always succeed.** A request is never refused because another one is in progress: it waits for the lock and then checks real data. There is a bounded retry for any unexpected transient error such as a serialization failure or deadlock.

```python
import datetime

import psycopg.errors as errors

_UTC = datetime.timezone.utc
_RETRYABLE = (
    errors.SerializationFailure,
    errors.DeadlockDetected,
    errors.LockNotAvailable,
)
_MAX_ATTEMPTS = 5


def migrate(db) -> None:
    db.execute(
        "CREATE INDEX IF NOT EXISTS bookings_room_starts_idx "
        "ON bookings (room_id, starts_at)"
    )


def _day_bounds(ts: datetime.datetime):
    d = ts.astimezone(_UTC).date()
    start = datetime.datetime(d.year, d.month, d.day, tzinfo=_UTC)
    return start, start + datetime.timedelta(days=1)


def _lock_room(db, room_id: int) -> None:
    db.execute(
        "SELECT id FROM rooms WHERE id = %s FOR NO KEY UPDATE",
        (room_id,),
    )


def _fits(db, room_id, starts_at, ends_at, exclude_id) -> bool:
    # Overlap check (half-open intervals), excluding the booking being moved.
    rows = db.execute(
        "SELECT 1 FROM bookings "
        "WHERE room_id = %(r)s AND starts_at < %(e)s AND ends_at > %(s)s "
        "AND id <> %(x)s LIMIT 1",
        {"r": room_id, "s": starts_at, "e": ends_at, "x": exclude_id},
    )
    if rows:
        return False
    day_start, day_end = _day_bounds(starts_at)
    rows = db.execute(
        "SELECT count(*) FROM bookings "
        "WHERE room_id = %(r)s AND starts_at >= %(ds)s AND starts_at < %(de)s "
        "AND id <> %(x)s",
        {"r": room_id, "ds": day_start, "de": day_end, "x": exclude_id},
    )
    return rows[0][0] < 3


def _safe_rollback(db) -> None:
    try:
        db.rollback()
    except Exception:
        pass


def book(db, room_id: int, starts_at: datetime.datetime,
         ends_at: datetime.datetime) -> bool:
    last_exc = None
    for _ in range(_MAX_ATTEMPTS):
        try:
            db.begin("read committed")
            _lock_room(db, room_id)
            if not _fits(db, room_id, starts_at, ends_at, -1):
                db.rollback()
                return False
            db.execute(
                "INSERT INTO bookings (room_id, starts_at, ends_at) "
                "VALUES (%s, %s, %s)",
                (room_id, starts_at, ends_at),
            )
            db.commit()
            return True
        except _RETRYABLE as exc:
            last_exc = exc
            _safe_rollback(db)
        except Exception:
            _safe_rollback(db)
            raise
    raise last_exc


def reschedule(db, booking_id: int, new_starts_at: datetime.datetime,
               new_ends_at: datetime.datetime) -> bool:
    last_exc = None
    for _ in range(_MAX_ATTEMPTS):
        try:
            db.begin("read committed")
            rows = db.execute(
                "SELECT room_id FROM bookings WHERE id = %s", (booking_id,)
            )
            if not rows:
                db.rollback()
                return False
            room_id = rows[0][0]
            _lock_room(db, room_id)
            # Re-check existence under the room lock (it may have been cancelled).
            rows = db.execute(
                "SELECT id FROM bookings WHERE id = %s FOR UPDATE",
                (booking_id,),
            )
            if not rows:
                db.rollback()
                return False
            if not _fits(db, room_id, new_starts_at, new_ends_at, booking_id):
                db.rollback()
                return False
            db.execute(
                "UPDATE bookings SET starts_at = %s, ends_at = %s WHERE id = %s",
                (new_starts_at, new_ends_at, booking_id),
            )
            db.commit()
            return True
        except _RETRYABLE as exc:
            last_exc = exc
            _safe_rollback(db)
        except Exception:
            _safe_rollback(db)
            raise
    raise last_exc


def cancel(db, booking_id: int) -> bool:
    last_exc = None
    for _ in range(_MAX_ATTEMPTS):
        try:
            rows = db.execute(
                "DELETE FROM bookings WHERE id = %s RETURNING id", (booking_id,)
            )
            return bool(rows)
        except _RETRYABLE as exc:
            last_exc = exc
    raise last_exc
```
