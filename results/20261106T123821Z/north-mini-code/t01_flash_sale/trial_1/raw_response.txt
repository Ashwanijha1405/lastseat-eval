import psycopg.errors

def purchase(db, product_id: int, qty: int) -> bool:
    """
    Atomically attempt to purchase `qty` units of product `product_id`.
    Returns True if the stock was sufficient and the order recorded,
    False otherwise. Retries a bounded number of times on serialization
    errors or deadlocks.
    """
    max_attempts = 5
    for _ in range(max_attempts):
        try:
            db.begin(isolation="read committed")

            # Try to decrement stock only if enough is present.
            # The UPDATE ... RETURNING returns the product id if the row was updated.
            updated = db.execute(
                "UPDATE products SET stock = stock - %(qty)s "
                "WHERE id = %(product_id)s AND stock >= %(qty)s "
                "RETURNING id",
                {"product_id": product_id, "qty": qty}
            )
            if not updated:          # insufficient stock
                db.rollback()
                return False

            # Record the order for the consumed units.
            db.execute(
                "INSERT INTO orders (product_id, qty) VALUES (%(product_id)s, %(qty)s)",
                {"product_id": product_id, "qty": qty}
            )

            db.commit()
            return True

        except psycopg.errors.SerializationFailure:
            db.rollback()
            continue
        except psycopg.errors.DeadlockDetected:
            db.rollback()
            continue
        except Exception:
            # For any other error, rollback and propagate it.
            db.rollback()
            raise

    # All retries exhausted – abort and signal an error.
    db.rollback()
    raise RuntimeError("purchase retries exhausted")
