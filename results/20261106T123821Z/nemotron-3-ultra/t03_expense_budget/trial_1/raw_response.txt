import psycopg.errors

def approve_expense(db, expense_id: int) -> bool:
    """
    Approve an expense if it is pending and the department's total approved amount
    stays within budget. Returns True on success, False otherwise.
    """
    max_attempts = 5
    for _ in range(max_attempts):
        try:
            db.begin()

            # Immutable fields: department_id and amount never change.
            rows = db.execute(
                "SELECT department_id, amount FROM expenses WHERE id = %s",
                (expense_id,)
            )
            if not rows:
                db.rollback()
                return False
            dept_id, amount = rows[0]

            # Lock the department row to serialize budget checks for this department.
            rows = db.execute(
                "SELECT id, budget FROM departments WHERE id = %s FOR UPDATE",
                (dept_id,)
            )
            if not rows:
                db.rollback()
                return False
            _, budget = rows[0]

            # Lock the expense row to inspect and possibly update its status.
            rows = db.execute(
                "SELECT id, status, amount FROM expenses WHERE id = %s FOR UPDATE",
                (expense_id,)
            )
            if not rows:
                db.rollback()
                return False
            _, status, amount = rows[0]

            if status != 'pending':
                db.rollback()
                return False

            # Current sum of already approved expenses in this department.
            rows = db.execute(
                "SELECT COALESCE(SUM(amount), 0) "
                "FROM expenses WHERE department_id = %s AND status = 'approved'",
                (dept_id,)
            )
            approved_sum = rows[0][0]

            if approved_sum + amount <= budget:
                db.execute(
                    "UPDATE expenses SET status = 'approved' WHERE id = %s",
                    (expense_id,)
                )
                db.commit()
                return True
            else:
                db.rollback()
                return False

        except (psycopg.errors.DeadlockDetected, psycopg.errors.SerializationFailure):
            # Transient concurrency error – roll back and retry.
            db.rollback()
            continue
        except psycopg.errors.DatabaseError:
            # Any other database error – treat as a refusal.
            db.rollback()
            return False

    # Extremely unlikely: all retry attempts failed.
    return False


def migrate(db) -> None:
    """
    Optional migration: create an index to speed up the budget-check query.
    """
    db.execute(
        "CREATE INDEX IF NOT EXISTS expenses_dept_status_idx "
        "ON expenses (department_id, status)"
    )
