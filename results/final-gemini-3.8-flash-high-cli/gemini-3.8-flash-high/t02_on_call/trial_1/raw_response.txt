from __future__ import annotations

from psycopg import errors


def migrate(db) -> None:
    db.execute(
        "CREATE INDEX IF NOT EXISTS idx_doctors_shift_on_call ON doctors (shift_id, id) WHERE on_call = true"
    )


def go_off_call(db, doctor_id: int) -> bool:
    max_retries = 3
    for attempt in range(max_retries):
        try:
            db.begin("read committed")
            rows = db.execute(
                "SELECT shift_id FROM doctors WHERE id = %s", (doctor_id,)
            )
            if not rows:
                db.rollback()
                return False
            shift_id = rows[0][0]

            # Serialize operations within the same shift to prevent race conditions
            # while allowing concurrent operations across different shifts.
            db.execute("SELECT pg_advisory_xact_lock(%s::bigint)", (shift_id,))

            # In read committed mode, this query sees the latest committed state
            # after acquiring the advisory lock. We atomically verify another doctor
            # is on call and update this doctor's status.
            updated = db.execute(
                """
                UPDATE doctors
                SET on_call = false
                WHERE id = %(doctor_id)s
                  AND on_call = true
                  AND EXISTS (
                      SELECT 1 FROM doctors
                      WHERE shift_id = %(shift_id)s
                        AND on_call = true
                        AND id != %(doctor_id)s
                  )
                RETURNING id
                """,
                {"doctor_id": doctor_id, "shift_id": shift_id},
            )
            db.commit()
            return bool(updated)
        except errors.Error:
            try:
                db.rollback()
            except Exception:
                pass
            if attempt == max_retries - 1:
                raise

    return False
