from __future__ import annotations

from datetime import datetime
from psycopg import errors


def migrate(db) -> None:
    db.execute(
        "CREATE INDEX IF NOT EXISTS idx_bookings_room_starts ON bookings (room_id, starts_at)"
    )


def book(db, room_id: int, starts_at: datetime, ends_at: datetime) -> bool:
    for _ in range(10):
        try:
            db.begin()
            db.execute(
                "SELECT pg_advisory_xact_lock(%(room_id)s::bigint)",
                {"room_id": room_id},
            )
            inserted = db.execute(
                """
                INSERT INTO bookings (room_id, starts_at, ends_at)
                SELECT %(room_id)s, %(starts_at)s, %(ends_at)s
                WHERE (
                    SELECT count(*)
                    FROM bookings
                    WHERE room_id = %(room_id)s
                      AND (starts_at AT TIME ZONE 'UTC')::date = (%(starts_at)s AT TIME ZONE 'UTC')::date
                ) < 3
                AND NOT EXISTS (
                    SELECT 1
                    FROM bookings
                    WHERE room_id = %(room_id)s
                      AND starts_at < %(ends_at)s
                      AND ends_at > %(starts_at)s
                )
                RETURNING id
                """,
                {
                    "room_id": room_id,
                    "starts_at": starts_at,
                    "ends_at": ends_at,
                },
            )
            if inserted:
                db.commit()
                return True
            db.rollback()
            return False
        except (errors.DatabaseError, errors.OperationalError):
            try:
                db.rollback()
            except Exception:
                pass
            continue
    return False


def reschedule(
    db, booking_id: int, new_starts_at: datetime, new_ends_at: datetime
) -> bool:
    for _ in range(10):
        try:
            db.begin()
            rows = db.execute(
                "SELECT room_id FROM bookings WHERE id = %(booking_id)s",
                {"booking_id": booking_id},
            )
            if not rows:
                db.rollback()
                return False
            room_id = rows[0][0]

            db.execute(
                "SELECT pg_advisory_xact_lock(%(room_id)s::bigint)",
                {"room_id": room_id},
            )
            updated = db.execute(
                """
                UPDATE bookings
                SET starts_at = %(new_starts_at)s,
                    ends_at   = %(new_ends_at)s
                WHERE id = %(booking_id)s
                  AND (
                      SELECT count(*)
                      FROM bookings
                      WHERE room_id = %(room_id)s
                        AND id != %(booking_id)s
                        AND (starts_at AT TIME ZONE 'UTC')::date = (%(new_starts_at)s AT TIME ZONE 'UTC')::date
                  ) < 3
                  AND NOT EXISTS (
                      SELECT 1
                      FROM bookings
                      WHERE room_id = %(room_id)s
                        AND id != %(booking_id)s
                        AND starts_at < %(new_ends_at)s
                        AND ends_at > %(new_starts_at)s
                  )
                RETURNING id
                """,
                {
                    "booking_id": booking_id,
                    "room_id": room_id,
                    "new_starts_at": new_starts_at,
                    "new_ends_at": new_ends_at,
                },
            )
            if updated:
                db.commit()
                return True
            db.rollback()
            return False
        except (errors.DatabaseError, errors.OperationalError):
            try:
                db.rollback()
            except Exception:
                pass
            continue
    return False


def cancel(db, booking_id: int) -> bool:
    for _ in range(10):
        try:
            db.begin()
            rows = db.execute(
                "SELECT room_id FROM bookings WHERE id = %(booking_id)s",
                {"booking_id": booking_id},
            )
            if not rows:
                db.rollback()
                return False
            room_id = rows[0][0]

            db.execute(
                "SELECT pg_advisory_xact_lock(%(room_id)s::bigint)",
                {"room_id": room_id},
            )
            deleted = db.execute(
                "DELETE FROM bookings WHERE id = %(booking_id)s RETURNING id",
                {"booking_id": booking_id},
            )
            if deleted:
                db.commit()
                return True
            db.rollback()
            return False
        except (errors.DatabaseError, errors.OperationalError):
            try:
                db.rollback()
            except Exception:
                pass
            continue
    return False
