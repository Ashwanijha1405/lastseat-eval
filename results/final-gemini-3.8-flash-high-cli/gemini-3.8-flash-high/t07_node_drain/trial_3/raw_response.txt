An implementation of `drain_node` and schema migration using PostgreSQL 16.

### Concurrency and Strategy

1. **Transaction-level serialization**: Draining reduces active replicas monotonically. To guarantee that concurrent drains never leave any shard without active replicas (Rule 1) and that all mutually valid drains succeed under concurrency (Rule 2), each drain transaction acquires a transaction-level exclusive advisory lock (`pg_advisory_xact_lock`). This serializes concurrent drain operations without deadlocks or table-level locks.
2. **Fresh MVCC snapshot**: Because `read committed` isolation takes a fresh statement snapshot after acquiring the advisory lock, each request observes all previously committed drains.
3. **Optimized indexes**: In `migrate(db)`, partial indexes on active replicas are created to make both the safety check and replica update fast index-only operations.
4. **All-or-nothing guarantee**: If any shard that has an active replica on the target node lacks another active replica on a different node, the transaction commits without modifying any rows and returns `False`. Otherwise, it deactivates the node's replicas and returns `True`.

```python
from __future__ import annotations

from psycopg import errors

# Unique 64-bit advisory lock key to serialize drain evaluations
_DRAIN_LOCK_ID = 6001150428613362770


def migrate(db) -> None:
    db.execute(
        "CREATE INDEX IF NOT EXISTS idx_replicas_node_active "
        "ON replicas (node_id, shard_id) WHERE active = true"
    )
    db.execute(
        "CREATE INDEX IF NOT EXISTS idx_replicas_shard_active "
        "ON replicas (shard_id, node_id) WHERE active = true"
    )


def drain_node(db, node_id: str) -> bool:
    for _ in range(12):
        try:
            db.begin()
            db.execute(
                "SELECT pg_advisory_xact_lock(%(lock_id)s)",
                {"lock_id": _DRAIN_LOCK_ID},
            )

            # Check if any shard with an active replica on this node
            # would be left with no active replica on any other node.
            blocked = db.execute(
                """
                SELECT 1
                FROM replicas cs
                WHERE cs.node_id = %(node_id)s
                  AND cs.active = true
                  AND NOT EXISTS (
                      SELECT 1
                      FROM replicas r
                      WHERE r.shard_id = cs.shard_id
                        AND r.node_id != %(node_id)s
                        AND r.active = true
                  )
                LIMIT 1
                """,
                {"node_id": node_id},
            )

            if blocked:
                db.commit()
                return False

            db.execute(
                """
                UPDATE replicas
                SET active = false
                WHERE node_id = %(node_id)s
                  AND active = true
                """,
                {"node_id": node_id},
            )
            db.commit()
            return True
        except (errors.SerializationFailure, errors.DeadlockDetected):
            try:
                db.rollback()
            except Exception:
                pass
            continue
        except Exception:
            try:
                db.rollback()
            except Exception:
                pass
            raise

    return False
```
