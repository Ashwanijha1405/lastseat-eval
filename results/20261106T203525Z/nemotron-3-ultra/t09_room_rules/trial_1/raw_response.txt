import datetime
import psycopg.errors

def _retry_transaction(db, max_retries=3):
    """Context manager-like helper to run a transaction with retries on serialization/deadlock errors.
    Yields a function to execute statements; the caller must call commit or rollback via db.
    """
    for attempt in range(max_retries):
        try:
            db.begin()  # default isolation: read committed
            db.execute("SET LOCAL TIME ZONE 'UTC'")
            yield
            db.commit()
            return
        except (psycopg.errors.SerializationFailure, psycopg.errors.DeadlockDetected):
            db.rollback()
            if attempt == max_retries - 1:
                # After exhausting retries, treat as failure (should be extremely rare with our locking)
                return False
        except Exception:
            db.rollback()
            raise
    return False  # only reached if all retries failed

def reschedule(db, booking_id: int, new_starts_at: datetime.datetime, new_ends_at: datetime.datetime) -> bool:
    max_retries = 3
    for attempt in range(max_retries):
        try:
            db.begin()
            db.execute("SET LOCAL TIME ZONE 'UTC'")

            # Get the room_id for this booking
            res = db.execute("SELECT room_id FROM bookings WHERE id = %s", (booking_id,))
            if not res:
                db.rollback()
                return False
            room_id = res[0][0]

            # Lock the room to serialize all changes for this room
            db.execute("SELECT 1 FROM rooms WHERE id = %s FOR UPDATE", (room_id,))

            # Lock the booking row and verify it still exists
            res = db.execute("SELECT id FROM bookings WHERE id = %s FOR UPDATE", (booking_id,))
            if not res:
                db.rollback()
                return False

            # Get the old start time to compute its day
            res = db.execute("SELECT starts_at FROM bookings WHERE id = %s", (booking_id,))
            old_starts_at = res[0][0]
            old_day = old_starts_at.astimezone(datetime.timezone.utc).date()
            new_day = new_starts_at.astimezone(datetime.timezone.utc).date()

            # Check daily limit on the new day
            res = db.execute(
                "SELECT COUNT(*) FROM bookings WHERE room_id = %s AND starts_at::date = %s",
                (room_id, new_day),
            )
            count = res[0][0] if res else 0
            if old_day == new_day:
                if count > 3:
                    db.rollback()
                    return False
            else:
                if count + 1 > 3:
                    db.rollback()
                    return False

            # Check overlap with other bookings of the same room
            res = db.execute(
                "SELECT 1 FROM bookings WHERE room_id = %s AND id != %s AND starts_at < %s AND ends_at > %s",
                (room_id, booking_id, new_ends_at, new_starts_at),
            )
            if res:
                db.rollback()
                return False

            # Update the booking
            db.execute(
                "UPDATE bookings SET starts_at = %s, ends_at = %s WHERE id = %s",
                (new_starts_at, new_ends_at, booking_id),
            )
            db.commit()
            return True
        except (psycopg.errors.SerializationFailure, psycopg.errors.DeadlockDetected):
            db.rollback()
            if attempt == max_retries - 1:
                return False
        except Exception:
            db.rollback()
            raise
    return False


def cancel(db, booking_id: int) -> bool:
    max_retries = 3
    for attempt in range(max_retries):
        try:
            db.begin()
            db.execute("SET LOCAL TIME ZONE 'UTC'")

            # Get the room_id
            res = db.execute("SELECT room_id FROM bookings WHERE id = %s", (booking_id,))
            if not res:
                db.rollback()
                return False
            room_id = res[0][0]

            # Lock the room
            db.execute("SELECT 1 FROM rooms WHERE id = %s FOR UPDATE", (room_id,))

            # Delete the booking and check if it existed
            res = db.execute("DELETE FROM bookings WHERE id = %s RETURNING id", (booking_id,))
            db.commit()
            return bool(res)
        except (psycopg.errors.SerializationFailure, psycopg.errors.DeadlockDetected):
            db.rollback()
            if attempt == max_retries - 1:
                return False
        except Exception:
            db.rollback()
            raise
    return False
