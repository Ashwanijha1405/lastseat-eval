```python
from datetime import datetime, timedelta, timezone

_EPOCH = datetime(1970, 1, 1, tzinfo=timezone.utc)


def migrate(db) -> None:
    db.execute(
        "CREATE INDEX IF NOT EXISTS bookings_room_starts_idx "
        "ON bookings (room_id, starts_at)"
    )


def _day_bounds(ts: datetime):
    t = ts.astimezone(timezone.utc)
    ds = t.replace(hour=0, minute=0, second=0, microsecond=0)
    de = ds + timedelta(days=1)
    key = (ds - _EPOCH).days
    return ds, de, key


def book(db, room_id: int, starts_at: datetime, ends_at: datetime) -> bool:
    ds, de, key = _day_bounds(starts_at)
    db.begin()
    try:
        # Per (room, day) lock: only requests for the same room and day serialize.
        db.execute("SELECT pg_advisory_xact_lock(%s, %s)", (room_id, key))
        overlap = db.execute(
            "SELECT 1 FROM bookings WHERE room_id = %s "
            "AND starts_at < %s AND ends_at > %s LIMIT 1",
            (room_id, ends_at, starts_at),
        )
        if overlap:
            db.rollback()
            return False
        cnt = db.execute(
            "SELECT count(*) FROM bookings WHERE room_id = %s "
            "AND starts_at >= %s AND starts_at < %s",
            (room_id, ds, de),
        )[0][0]
        if cnt >= 3:
            db.rollback()
            return False
        db.execute(
            "INSERT INTO bookings (room_id, starts_at, ends_at) VALUES (%s, %s, %s)",
            (room_id, starts_at, ends_at),
        )
        db.commit()
        return True
    except BaseException:
        try:
            db.rollback()
        except Exception:
            pass
        raise


def reschedule(db, booking_id: int, new_starts_at: datetime, new_ends_at: datetime) -> bool:
    # The room of a booking never changes, so it can be read before locking.
    rows = db.execute("SELECT room_id FROM bookings WHERE id = %s", (booking_id,))
    if not rows:
        return False
    room_id = rows[0][0]
    ds, de, key = _day_bounds(new_starts_at)
    db.begin()
    try:
        # Only the destination (room, day) needs the lock: leaving the old day
        # can only reduce its count, so it never needs to block anyone.
        db.execute("SELECT pg_advisory_xact_lock(%s, %s)", (room_id, key))
        cur = db.execute(
            "SELECT id FROM bookings WHERE id = %s FOR UPDATE", (booking_id,)
        )
        if not cur:
            db.rollback()
            return False
        overlap = db.execute(
            "SELECT 1 FROM bookings WHERE room_id = %s AND id <> %s "
            "AND starts_at < %s AND ends_at > %s LIMIT 1",
            (room_id, booking_id, new_ends_at, new_starts_at),
        )
        if overlap:
            db.rollback()
            return False
        others = db.execute(
            "SELECT count(*) FROM bookings WHERE room_id = %s AND id <> %s "
            "AND starts_at >= %s AND starts_at < %s",
            (room_id, booking_id, ds, de),
        )[0][0]
        if others >= 3:
            db.rollback()
            return False
        db.execute(
            "UPDATE bookings SET starts_at = %s, ends_at = %s WHERE id = %s",
            (new_starts_at, new_ends_at, booking_id),
        )
        db.commit()
        return True
    except BaseException:
        try:
            db.rollback()
        except Exception:
            pass
        raise


def cancel(db, booking_id: int) -> bool:
    # Deleting only frees capacity, so a single autocommitted statement suffices.
    rows = db.execute(
        "DELETE FROM bookings WHERE id = %s RETURNING id", (booking_id,)
    )
    return bool(rows)
```

**Locking.** `book` and `reschedule` take a transaction-level advisory lock keyed on `(room_id, UTC day)`. Requests for different rooms or different days never wait on each other. Bookings never span days, so one lock covers both the overlap check and the 3-per-day count. Under read committed, each statement after the lock sees a fresh snapshot, so those checks are accurate.

**`reschedule`.** It locks only the destination room-day. Moving out of the old day can only lower its count, so that day needs no lock. This also means each request holds a single advisory lock, so two requests can't deadlock on each other's locks. The booking row is then locked with `FOR UPDATE`. A concurrent cancel is handled: it returns `False` if the row is gone. The overlap and count checks exclude the booking itself.

**`cancel`.** It is a single `DELETE ... RETURNING`. Deleting never breaks a rule, so it needs no advisory lock.

**Limitation.** Where the outcome depends on request order, the result follows the order in which requests take the lock. For example, a `book` into a full day that arrives just before a `cancel` on that day returns `False`.

**Not run.** I didn't run this against PostgreSQL. `migrate` only adds an index on `(room_id, starts_at)` and no constraints, so existing data can't make it fail.
