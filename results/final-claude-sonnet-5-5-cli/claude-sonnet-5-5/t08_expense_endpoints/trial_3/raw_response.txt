All three functions lock the department row with `SELECT ... FOR UPDATE`, so every change that affects a department's approved total is serialized per department. I haven't run this against a database.

- **Lock order:** `expenses.department_id` never changes, so it can be read without a lock. Each request then takes a single lock, on the department row, and cannot deadlock.
- **Fresh reads:** after the lock is held, the expense and the approved total are re-read. In read committed each statement gets a new snapshot, so these reads see everything committed by earlier lock holders.
- **Rule 1:** only the lock holder changes approved amounts, and it checks the limit against current data first, so the approved total never exceeds the budget.
- **Rule 2:** a request returns `False` only when it fails against the committed state it saw under the lock. Requests that would all succeed in some sequential order therefore don't fail just because they overlap.

```python
import psycopg.errors


def _lock_dept(db, dept_id):
    rows = db.execute(
        "SELECT budget FROM departments WHERE id = %s FOR UPDATE", (dept_id,)
    )
    return rows[0][0]


def _approved_total(db, dept_id):
    rows = db.execute(
        "SELECT COALESCE(SUM(amount), 0) FROM expenses "
        "WHERE department_id = %s AND status = 'approved'",
        (dept_id,),
    )
    return int(rows[0][0])


def approve_expense(db, expense_id: int) -> bool:
    rows = db.execute(
        "SELECT department_id FROM expenses WHERE id = %s", (expense_id,)
    )
    dept_id = rows[0][0]
    db.begin("read committed")
    try:
        budget = _lock_dept(db, dept_id)
        rows = db.execute(
            "SELECT status, amount FROM expenses WHERE id = %s", (expense_id,)
        )
        status, amount = rows[0]
        if status != "pending":
            db.rollback()
            return False
        total = _approved_total(db, dept_id)
        if total + amount > budget:
            db.rollback()
            return False
        db.execute(
            "UPDATE expenses SET status = 'approved' WHERE id = %s", (expense_id,)
        )
        db.commit()
        return True
    except Exception:
        try:
            db.rollback()
        except Exception:
            pass
        raise


def edit_expense(db, expense_id: int, new_amount: int) -> bool:
    rows = db.execute(
        "SELECT department_id FROM expenses WHERE id = %s", (expense_id,)
    )
    dept_id = rows[0][0]
    db.begin("read committed")
    try:
        budget = _lock_dept(db, dept_id)
        rows = db.execute(
            "SELECT status, amount FROM expenses WHERE id = %s", (expense_id,)
        )
        status, amount = rows[0]
        if status == "approved":
            total = _approved_total(db, dept_id)
            if total - amount + new_amount > budget:
                db.rollback()
                return False
        db.execute(
            "UPDATE expenses SET amount = %s WHERE id = %s", (new_amount, expense_id)
        )
        db.commit()
        return True
    except Exception:
        try:
            db.rollback()
        except Exception:
            pass
        raise


def change_budget(db, dept_id: int, new_budget: int) -> bool:
    db.begin("read committed")
    try:
        _lock_dept(db, dept_id)
        total = _approved_total(db, dept_id)
        if total > new_budget:
            db.rollback()
            return False
        db.execute(
            "UPDATE departments SET budget = %s WHERE id = %s", (new_budget, dept_id)
        )
        db.commit()
        return True
    except Exception:
        try:
            db.rollback()
        except Exception:
            pass
        raise
```
