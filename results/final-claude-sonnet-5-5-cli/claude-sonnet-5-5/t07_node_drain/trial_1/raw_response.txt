import psycopg.errors


def drain_node(db, node_id: str) -> bool:
    # Concurrent drains that touch a common shard are serialized by locking
    # the shard rows, always in ascending id order, so they cannot deadlock.
    # Drains never conflict unless they share a shard, so valid requests
    # are never refused because of concurrency.
    # Whether a set of drains all succeed does not depend on their order,
    # so serializing them in any order is safe.
    for _ in range(5):
        try:
            db.begin("read committed")
            db.execute(
                """
                SELECT s.id FROM shards s
                WHERE s.id IN (SELECT shard_id FROM replicas WHERE node_id = %(n)s)
                ORDER BY s.id
                FOR UPDATE
                """,
                {"n": node_id},
            )
            # In read committed, this statement gets a fresh snapshot taken
            # after the locks were acquired, so it sees all committed drains.
            bad = db.execute(
                """
                SELECT 1 FROM replicas r
                WHERE r.node_id = %(n)s AND r.active
                  AND NOT EXISTS (
                      SELECT 1 FROM replicas o
                      WHERE o.shard_id = r.shard_id
                        AND o.node_id <> %(n)s
                        AND o.active
                  )
                LIMIT 1
                """,
                {"n": node_id},
            )
            if bad:
                db.rollback()
                return False
            db.execute(
                "UPDATE replicas SET active = false WHERE node_id = %(n)s AND active",
                {"n": node_id},
            )
            db.commit()
            return True
        except (psycopg.errors.DeadlockDetected, psycopg.errors.SerializationFailure):
            db.rollback()
    return False
