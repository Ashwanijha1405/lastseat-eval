Concurrent drains can break rule 1 through write skew: two drains each see the other node's replica as still active and both proceed. To prevent this, each drain first locks the `shards` rows it touches, in ascending id order, so conflicting drains are serialized and can't deadlock. After taking the locks, it checks and updates under a fresh read-committed snapshot.

Drains that don't share a shard don't block each other. If a set of drains succeeds in some order, it succeeds in any order, so serializing them can't make a drain fail that would have succeeded. I haven't run this against a PostgreSQL server.

```python
import psycopg
import psycopg.errors


def drain_node(db, node_id: str) -> bool:
    for _ in range(5):
        try:
            db.begin("read committed")
            # Lock every shard this node has a replica of, in a fixed order.
            db.execute(
                """
                SELECT s.id
                FROM shards s
                JOIN replicas r ON r.shard_id = s.id
                WHERE r.node_id = %(n)s
                ORDER BY s.id
                FOR UPDATE OF s
                """,
                {"n": node_id},
            )
            # Fresh snapshot (read committed) after locks are held.
            bad = db.execute(
                """
                SELECT 1
                FROM replicas r
                WHERE r.node_id = %(n)s
                  AND r.active
                  AND NOT EXISTS (
                      SELECT 1 FROM replicas o
                      WHERE o.shard_id = r.shard_id
                        AND o.node_id <> r.node_id
                        AND o.active
                  )
                LIMIT 1
                """,
                {"n": node_id},
            )
            if bad:
                db.rollback()
                return False
            db.execute(
                "UPDATE replicas SET active = false WHERE node_id = %(n)s AND active",
                {"n": node_id},
            )
            db.commit()
            return True
        except (psycopg.errors.DeadlockDetected, psycopg.errors.SerializationFailure):
            db.rollback()
            continue
    return False
```
