import datetime
import psycopg.errors

def migrate(db) -> None:
    # Create a table to track booking counts per room per UTC day.
    db.execute("""
        CREATE TABLE room_day (
            room_id     int NOT NULL,
            day         date NOT NULL,
            booking_count int NOT NULL DEFAULT 0,
            PRIMARY KEY (room_id, day)
        )
    """)
    # Populate it with current counts from existing bookings.
    db.execute("""
        INSERT INTO room_day (room_id, day, booking_count)
        SELECT room_id, starts_at::date, count(*)
        FROM bookings
        GROUP BY room_id, starts_at::date
    """)


def _utc_date(dt: datetime.datetime) -> datetime.date:
    """Convert a timezone-aware datetime to its UTC calendar date."""
    return dt.astimezone(datetime.timezone.utc).date()


def book(db, room_id: int, starts_at: datetime.datetime, ends_at: datetime.datetime) -> bool:
    day = _utc_date(starts_at)
    for _ in range(3):  # bounded retry on serialization/deadlock
        try:
            db.begin()
            # Ensure the room_day row exists and lock it.
            db.execute(
                "INSERT INTO room_day (room_id, day, booking_count) VALUES (%(room_id)s, %(day)s, 0) "
                "ON CONFLICT (room_id, day) DO NOTHING",
                {"room_id": room_id, "day": day}
            )
            rows = db.execute(
                "SELECT booking_count FROM room_day WHERE room_id = %(room_id)s AND day = %(day)s FOR UPDATE",
                {"room_id": room_id, "day": day}
            )
            booking_count = rows[0][0]

            # Check for overlapping bookings on the same room.
            rows = db.execute(
                "SELECT 1 FROM bookings WHERE room_id = %(room_id)s "
                "AND starts_at < %(ends_at)s AND ends_at > %(starts_at)s LIMIT 1",
                {"room_id": room_id, "starts_at": starts_at, "ends_at": ends_at}
            )
            if rows or booking_count >= 3:
                db.rollback()
                return False

            # Insert the new booking and increment the day's count.
            db.execute(
                "INSERT INTO bookings (room_id, starts_at, ends_at) VALUES (%(room_id)s, %(starts_at)s, %(ends_at)s)",
                {"room_id": room_id, "starts_at": starts_at, "ends_at": ends_at}
            )
            db.execute(
                "UPDATE room_day SET booking_count = booking_count + 1 WHERE room_id = %(room_id)s AND day = %(day)s",
                {"room_id": room_id, "day": day}
            )
            db.commit()
            return True
        except (psycopg.errors.SerializationFailure, psycopg.errors.DeadlockDetected):
            db.rollback()
            continue
        except Exception:
            db.rollback()
            return False
    return False


def reschedule(db, booking_id: int, new_starts_at: datetime.datetime, new_ends_at: datetime.datetime) -> bool:
    new_day = _utc_date(new_starts_at)
    for _ in range(3):
        try:
            db.begin()
            # Lock the booking row first.
            rows = db.execute(
                "SELECT room_id, starts_at FROM bookings WHERE id = %(booking_id)s FOR UPDATE",
                {"booking_id": booking_id}
            )
            if not rows:
                db.rollback()
                return False
            room_id, old_starts_at = rows[0]
            old_day = _utc_date(old_starts_at)

            # Lock the affected room_day rows in a consistent order (by day).
            days = sorted({old_day, new_day})
            counts = {}
            for day in days:
                db.execute(
                    "INSERT INTO room_day (room_id, day, booking_count) VALUES (%(room_id)s, %(day)s, 0) "
                    "ON CONFLICT (room_id, day) DO NOTHING",
                    {"room_id": room_id, "day": day}
                )
                rows = db.execute(
                    "SELECT booking_count FROM room_day WHERE room_id = %(room_id)s AND day = %(day)s FOR UPDATE",
                    {"room_id": room_id, "day": day}
                )
                counts[day] = rows[0][0]

            # Check overlap with *other* bookings on the new day.
            rows = db.execute(
                "SELECT 1 FROM bookings WHERE room_id = %(room_id)s AND id != %(booking_id)s "
                "AND starts_at < %(new_ends_at)s AND ends_at > %(new_starts_at)s LIMIT 1",
                {"room_id": room_id, "booking_id": booking_id,
                 "new_starts_at": new_starts_at, "new_ends_at": new_ends_at}
            )
            if rows:
                db.rollback()
                return False

            # If moving to a different day, ensure the new day has capacity.
            if new_day != old_day and counts[new_day] >= 3:
                db.rollback()
                return False

            # Update the booking times.
            db.execute(
                "UPDATE bookings SET starts_at = %(new_starts_at)s, ends_at = %(new_ends_at)s WHERE id = %(booking_id)s",
                {"booking_id": booking_id, "new_starts_at": new_starts_at, "new_ends_at": new_ends_at}
            )

            # Adjust day counts if the day changed.
            if new_day != old_day:
                db.execute(
                    "UPDATE room_day SET booking_count = booking_count - 1 WHERE room_id = %(room_id)s AND day = %(old_day)s",
                    {"room_id": room_id, "old_day": old_day}
                )
                db.execute(
                    "UPDATE room_day SET booking_count = booking_count + 1 WHERE room_id = %(room_id)s AND day = %(new_day)s",
                    {"room_id": room_id, "new_day": new_day}
                )
            db.commit()
            return True
        except (psycopg.errors.SerializationFailure, psycopg.errors.DeadlockDetected):
            db.rollback()
            continue
        except Exception:
            db.rollback()
            return False
    return False


def cancel(db, booking_id: int) -> bool:
    for _ in range(3):
        try:
            db.begin()
            # Lock the booking row.
            rows = db.execute(
                "SELECT room_id, starts_at FROM bookings WHERE id = %(booking_id)s FOR UPDATE",
                {"booking_id": booking_id}
            )
            if not rows:
                db.rollback()
                return False
            room_id, starts_at = rows[0]
            day = _utc_date(starts_at)

            # Lock the corresponding room_day row.
            db.execute(
                "INSERT INTO room_day (room_id, day, booking_count) VALUES (%(room_id)s, %(day)s, 0) "
                "ON CONFLICT (room_id, day) DO NOTHING",
                {"room_id": room_id, "day": day}
            )
            db.execute(
                "SELECT booking_count FROM room_day WHERE room_id = %(room_id)s AND day = %(day)s FOR UPDATE",
                {"room_id": room_id, "day": day}
            )

            # Delete the booking and decrement the day's count.
            db.execute(
                "DELETE FROM bookings WHERE id = %(booking_id)s",
                {"booking_id": booking_id}
            )
            db.execute(
                "UPDATE room_day SET booking_count = booking_count - 1 WHERE room_id = %(room_id)s AND day = %(day)s",
                {"room_id": room_id, "day": day}
            )
            db.commit()
            return True
        except (psycopg.errors.SerializationFailure, psycopg.errors.DeadlockDetected):
            db.rollback()
            continue
        except Exception:
            db.rollback()
            return False
    return False
