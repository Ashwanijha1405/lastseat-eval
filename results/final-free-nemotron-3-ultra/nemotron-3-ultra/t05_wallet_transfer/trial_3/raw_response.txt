import psycopg.errors

def migrate(db) -> None:
    # No schema changes needed; the existing accounts table with primary key on id is sufficient.
    pass

def transfer(db, from_id: int, to_id: int, amount: int) -> bool:
    """
    Perform a transfer from `from_id` to `to_id` of `amount` currency units.
    Returns True if the transfer succeeded, False if the sender had insufficient funds.
    Uses a receiver-first update order to allow opposite-direction transfers to succeed
    concurrently, with deadlock detection and retry.
    """
    max_attempts = 10  # well within the 60-call limit (4 calls per attempt)
    for _ in range(max_attempts):
        try:
            db.begin()  # default isolation level: read committed

            # 1. Credit the receiver (unconditional)
            db.execute(
                "UPDATE accounts SET balance = balance + %(amount)s WHERE id = %(to_id)s",
                {"amount": amount, "to_id": to_id}
            )

            # 2. Debit the sender, but only if the current balance is sufficient.
            #    The RETURNING clause lets us detect whether the update actually happened.
            result = db.execute(
                "UPDATE accounts SET balance = balance - %(amount)s "
                "WHERE id = %(from_id)s AND balance >= %(amount)s RETURNING 1",
                {"amount": amount, "from_id": from_id}
            )

            if not result:          # no row updated → insufficient balance
                db.rollback()
                return False

            db.commit()
            return True

        except psycopg.errors.DeadlockDetected:
            # Another transaction holds locks in the opposite order; retry.
            db.rollback()
            continue
        except psycopg.errors.SerializationFailure:
            # Should not occur in read committed, but retry defensively.
            db.rollback()
            continue
        except Exception:
            # Any other error: roll back and propagate (the test harness will treat this as a failure).
            db.rollback()
            raise

    # If we exhausted retries (extremely unlikely under normal contention), treat as failure.
    return False
