The fix is to serialise requests per shift by locking the shift row. Two doctors in the same shift can otherwise both see a colleague on call and both go off, leaving the shift empty. Locking the shift row makes each request's count run after the previous request has committed, so the final state matches some one-at-a-time ordering. Requests that would succeed in sequence also succeed here, with no retries and no refusals caused by concurrency. Every request locks the shift before touching any doctor row, so lock ordering is consistent and deadlocks can't occur. No `migrate` is needed.

```python
def go_off_call(db, doctor_id: int) -> bool:
    db.begin()  # "read committed"
    try:
        rows = db.execute(
            "SELECT shift_id FROM doctors WHERE id = %s",
            (doctor_id,),
        )
        if not rows:
            db.rollback()
            return False
        shift_id = rows[0][0]

        # Serialise off-call requests for this shift. Each request takes this
        # lock before reading or changing any doctor in the shift, so the count
        # below runs after every earlier request in the shift has committed.
        db.execute(
            "SELECT id FROM shifts WHERE id = %s FOR UPDATE",
            (shift_id,),
        )

        rows = db.execute(
            "SELECT count(*) FROM doctors "
            "WHERE shift_id = %s AND on_call AND id <> %s",
            (shift_id, doctor_id),
        )
        others_on_call = rows[0][0]

        if others_on_call <= 0:
            # This doctor is the only one on call: change nothing.
            db.commit()
            return False

        db.execute(
            "UPDATE doctors SET on_call = false WHERE id = %s",
            (doctor_id,),
        )
        db.commit()
        return True
    except Exception:
        db.rollback()
        raise
```

I haven't run this against PostgreSQL 16, so it's unverified.
